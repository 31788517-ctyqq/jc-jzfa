**`_switchTab` 改造**:

```javascript
function _switchTab(tab) {
  _currentTab = tab;
  document.querySelectorAll('.adm-tab').forEach(function (el) {
    el.classList.toggle('active', el.dataset.tab === tab);
  });
  const panel = document.getElementById('admPanel');
  if (!panel) return;
  panel.innerHTML = '<div class="adm-loading">加载中...</div>';

  switch (tab) {
    case 'users':    _renderUsersTab(panel);    break;
    case 'payments':  _renderPaymentsTab(panel); break;
    case 'referrals': _renderReferralsTab(panel); break;
    case 'system':    _renderSystemTab(panel);   break;
    // ── 3 个新增 Tab ──
    case 'pipeline':
      panel.innerHTML = '<div id="adminPipelineContent"><div class="page-skeleton">...</div></div>';
      _mod('admin-pipeline').then(m => m.loadAdminPipeline(panel));
      break;
    case 'compute':
      panel.innerHTML = '<div id="adminComputeContent"><div class="page-skeleton">...</div></div>';
      _mod('admin-compute').then(m => m.loadAdminCompute(panel));
      break;
    case 'overview':
      panel.innerHTML = '<div id="adminOverviewContent"><div class="page-skeleton">...</div></div>';
      _mod('admin-overview').then(m => m.loadAdminOverview(panel));
      break;
    default:
      panel.innerHTML = '<div class="adm-loading">未知页面</div>';
  }
}
```

## 8. 渲染快照检查 (data-snapshot.js)

### 8.1 工具库设计

```javascript
// preview/js/data-snapshot.js
// 轻量级渲染正确性检查 — 纯前端, 无外部依赖

const SNAPSHOT_SAMPLE_RATE = 0.1; // 10% 采样率

function initSnapshotCheck() {
  if (Math.random() > SNAPSHOT_SAMPLE_RATE) return; // 采样控制
  // 遍历所有 data-snapshot 属性元素
  document.querySelectorAll('[data-snapshot]').forEach(checkElement);
}

function checkElement(el) {
  const snapshotId = el.dataset.snapshot;
  const expected = el.dataset.expected;
  if (!snapshotId || !expected) return;

  // 序列化实际渲染内容 (不同区域不同策略)
  let actual = '';
  if (snapshotId.startsWith('odds-')) {
    // 赔率区域: 提取所有 .odds-cell 的文本
    actual = JSON.stringify(
      Array.from(el.querySelectorAll('.odds-cell')).map(c => c.textContent.trim())
    );
  } else if (snapshotId.startsWith('score-')) {
    // 比分区域: 提取比分文本
    actual = el.textContent.trim().replace(/\s+/g, ' ');
  } else {
    // 默认: 提取文本内容
    actual = el.textContent.trim();
  }

  if (actual !== expected) {
    console.warn(`[渲染差异] ${snapshotId}: 预期=${expected}, 实际=${actual}`);
    // 异步上报, 不阻塞
    snapshotReport(snapshotId, expected, actual);
  }
}

async function snapshotReport(snapshotId, expected, actual) {
  try {
    await api('render-check-report', {
      snapshotId,
      expected,
      actual,
      url: location.href,
      time: new Date().toISOString()
    });
  } catch (e) {
    // silent
  }
}

export { initSnapshotCheck };
```

#### M1: Tab 栏 4列 → 2行布局

**根因**: `.adm-tabs` 使用 `grid-template-columns: repeat(4, 1fr)`。  
**方案**: 改为自动换行的flex布局，或增加 `grid-template-areas` 分2行:

```css
.adm-tabs {
  /* grid: repeat(4) → flex wrap，自然分2行 */
  display: flex;
  flex-wrap: wrap;
  height: auto;           /* 取消固定高度 */
  gap: 6px;
  padding: 8px 10px;
}
.adm-tab {
  flex: 1 1 calc(25% - 6px);  /* 每行4个，第2行3个+空白 */
  min-width: 0;
  height: 52px;                /* 比原来矮一些，两行总高度 ~120px */
  font-size: 12px;
  gap: 4px;
}
/* 第2行第4个占位空白（7 Tab = 4+3） */
.adm-tab-placeholder {
  visibility: hidden;
  flex: 1 1 calc(25% - 6px);
  height: 0;
}
```

**注意**: Tab 9个时规律是:
- 4个Tab = 1行
- 5个 = 2行 (3+2)
- 6个 = 2行 (3+3)
- 7个 = 2行 (4+3)

第2行居中对齐使用 `justify-content: center`:

```css
.adm-tabs {
  display: flex;
  flex-wrap: wrap;
  justify-content: center;  /* 第2行居中 */
  gap: 6px;
  height: auto;
  padding: 8px 10px;
}
.adm-tab {
  flex: 0 1 calc(25% - 6px);  /* 0 1 防止过宽 */
  min-width: 70px;
  max-width: 120px;
  height: 48px;
}
```
